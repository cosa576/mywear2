  function addToCart(product, size = "M", quantity = 1) {
    if (!product) return;

    const qty = Math.max(1, Math.min(20, Math.floor(Number(quantity) || 1)));
    const key = `${product.id}-${size}`;

    const existing = state.cart.find(item => item.key === key);

    if (existing) {
      existing.quantity = Math.min(99, existing.quantity + qty);
    } else {
      state.cart.push({
        key,
        id: product.id,
        name: product.name,
        price: product.price,
        icon: product.icon,
        size,
        quantity: qty
      });
    }

    renderCart();
    toast("Producto añadido al carrito");
  }

    closeProduct();

     VISOR 3D CON THREE.JS
     Modelo 3D estilizado de una camiseta.
  ========================================= */

  let scene, camera, renderer, garmentGroup, garmentMaterial;
  let rotationY = 0;
  let dragging = false;
  let previousX = 0;
  let animationId = null;

  function createGarmentShape() {
    const shape = new THREE.Shape();

    shape.moveTo(-0.8, 1.35);
    shape.lineTo(-1.25, 1.55);
    shape.lineTo(-2.05, 1.05);
    shape.lineTo(-1.72, 0.25);
    shape.lineTo(-1.25, 0.48);
    shape.lineTo(-1.25, -1.65);
    shape.quadraticCurveTo(-1.25, -1.78, -1.1, -1.78);
    shape.lineTo(1.1, -1.78);
    shape.quadraticCurveTo(1.25, -1.78, 1.25, -1.65);
    shape.lineTo(1.25, 0.48);
    shape.lineTo(1.72, 0.25);
    shape.lineTo(2.05, 1.05);
    shape.lineTo(1.25, 1.55);
    shape.lineTo(0.8, 1.35);
    shape.lineTo(0.55, 1.18);
    shape.quadraticCurveTo(0, 0.92, -0.55, 1.18);
    shape.closePath();

    return shape;
  }

  function make3DModel() {
    if (!window.THREE) {
      $("mw3DLoading").textContent = "No se pudo cargar Three.js. Comprueba tu conexión.";
      return;
    }

    const container = $("mw3DCanvas");
    const width = container.clientWidth || 600;
    const height = container.clientHeight || 430;

    scene = new THREE.Scene();
    scene.background = new THREE.Color("#c9c9c7");

    camera = new THREE.PerspectiveCamera(35, width / height, 0.1, 100);
    camera.position.set(0, 0, 8);

    renderer = new THREE.WebGLRenderer({ antialias: true, alpha: false });
    renderer.setPixelRatio(Math.min(window.devicePixelRatio || 1, 2));
    renderer.setSize(width, height);
    container.appendChild(renderer.domElement);

    scene.add(new THREE.AmbientLight(0xffffff, 1.25));

    const light = new THREE.DirectionalLight(0xffffff, 2);
    light.position.set(-3, 5, 7);
    scene.add(light);

    const backLight = new THREE.DirectionalLight(0xffffff, .7);
    backLight.position.set(3, 2, -5);
    scene.add(backLight);

    garmentGroup = new THREE.Group();
    scene.add(garmentGroup);

    garmentMaterial = new THREE.MeshStandardMaterial({
      color: state.garmentColor,
      roughness: .88,
      metalness: 0
    });

    const geometry = new THREE.ExtrudeGeometry(createGarmentShape(), {
      depth: .28,
      bevelEnabled: true,
      bevelSegments: 3,
      steps: 1,
      bevelSize: .07,
      bevelThickness: .06
    });

    geometry.center();

    const body = new THREE.Mesh(geometry, garmentMaterial);
    garmentGroup.add(body);

    // Cuello delantero: aro tridimensional.
    const collarGeometry = new THREE.TorusGeometry(.43, .075, 12, 48);
    const collarMaterial = new THREE.MeshStandardMaterial({
      color: "#999999",
      roughness: .9
    });
    const collar = new THREE.Mesh(collarGeometry, collarMaterial);
    collar.position.set(0, 1.12, .20);
    collar.scale.set(1.15, .55, 1);
    garmentGroup.add(collar);

    // Costuras decorativas en la parte delantera.
    const seamMaterial = new THREE.LineBasicMaterial({
      color: "#777777",
      transparent: true,
      opacity: .65
    });

    const seamPoints = [
      new THREE.Vector3(-.98, -1.45, .19),
      new THREE.Vector3(-.98, .45, .19),
      new THREE.Vector3(-.75, 1.25, .19),
      new THREE.Vector3(0, 1.05, .19),
      new THREE.Vector3(.75, 1.25, .19),
      new THREE.Vector3(.98, .45, .19),
      new THREE.Vector3(.98, -1.45, .19)
    ];

    const seamGeometry = new THREE.BufferGeometry().setFromPoints(seamPoints);
    garmentGroup.add(new THREE.Line(seamGeometry, seamMaterial));

    garmentGroup.rotation.y = rotationY;
    garmentGroup.rotation.x = -.06;

    $("mw3DLoading").style.display = "none";

    renderer.domElement.addEventListener("pointerdown", event => {
      dragging = true;
      previousX = event.clientX;
      renderer.domElement.setPointerCapture(event.pointerId);
    });

    renderer.domElement.addEventListener("pointermove", event => {
      if (!dragging) return;

      const difference = event.clientX - previousX;
      rotationY += difference * .012;
      previousX = event.clientX;
      garmentGroup.rotation.y = rotationY;
      update3DAngle();
    });

    renderer.domElement.addEventListener("pointerup", () => {
      dragging = false;
    });

    renderer.domElement.addEventListener("pointercancel", () => {
      dragging = false;
    });

    renderer.domElement.addEventListener("wheel", event => {
      event.preventDefault();
      camera.position.z = Math.max(5, Math.min(11, camera.position.z + event.deltaY * .006));
    }, { passive: false });

    function animate() {
      animationId = requestAnimationFrame(animate);
      renderer.render(scene, camera);
    }

    animate();

    if (window.ResizeObserver) {
      const resizeObserver = new ResizeObserver(resize3D);
      resizeObserver.observe(container);
    } else {
      window.addEventListener("resize", resize3D);
    }
  }

  function resize3D() {
    if (!renderer || !camera) return;

    const container = $("mw3DCanvas");
    const width = container.clientWidth;
    const height = container.clientHeight;

    if (!width || !height) return;

    camera.aspect = width / height;
    camera.updateProjectionMatrix();
    renderer.setSize(width, height);
  }

  function update3DAngle() {
    let degrees = ((rotationY * 180 / Math.PI) % 360 + 360) % 360;
    let label = "Vista frontal";

    if (degrees >= 45 && degrees < 135) label = "Vista lateral";
    else if (degrees >= 135 && degrees < 225) label = "Vista posterior";
    else if (degrees >= 225 && degrees < 315) label = "Vista lateral";

    $("mw3DAngle").textContent = label;
  }

  function update3DColor(color) {
    if (garmentMaterial) garmentMaterial.color.set(color);
  }

  $("mw3DColor").addEventListener("input", event => {
    update3DColor(event.target.value);
    state.garmentColor = event.target.value;
    $("mwPreviewShirt").style.backgroundColor = state.garmentColor;
  });

  $("mw3DLeft").addEventListener("click", () => {
    rotationY -= Math.PI / 4;
    if (garmentGroup) garmentGroup.rotation.y = rotationY;
    update3DAngle();
  });

  $("mw3DRight").addEventListener("click", () => {
    rotationY += Math.PI / 4;
    if (garmentGroup) garmentGroup.rotation.y = rotationY;
    update3DAngle();
  });

  $("mw3DReset").addEventListener("click", () => {
    rotationY = 0;
    if (garmentGroup) {
      garmentGroup.rotation.y = 0;
      garmentGroup.rotation.x = -.06;
    }
    if (camera) camera.position.set(0, 0, 8);
    update3DAngle();
  });

  document.addEventListener("keydown", event => {
    if (event.key === "Escape") {
      if (!$("mwProductModal").hidden) closeProduct();
      if ($("mwCartDrawer").classList.contains("is-open")) closeCart();
    }
  });
